# DocScanner OCR & RAG Intelligence
import pytesseract
from rag_engine import ContextRetriever

class DocScanner:
    def __init__(self, file_path: str):
        self.doc_path = file_path
        self.rag = ContextRetriever()

    def extract_and_query(self, query: str) -> str:
        text = pytesseract.image_to_string(self.doc_path)
        return self.rag.generate_answer(text, query)
